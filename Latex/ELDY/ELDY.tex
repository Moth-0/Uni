\begin{align*}

The gradient is used to calculate the divergence of a vector with: 
\begin{align*}

    = \left(\frac{\dd v_z}{\dd y} - \frac{\dd v_y}{\dd z}\right)\hat{x} &+ \left(\frac{\dd v_x}{\dd z} - \frac{\dd v_z}{\dd x}\right)\hat{y} + \left(\frac{\dd v_y}{\dd x} - \frac{\dd v_x}{\dd y}\right)\hat{z}. \eq{1.41}

The product rules for the gradient are: 
\begin{align*}
    &\nabla (fg) = f\nabla g + g\nabla f \\
    &\nabla (\bar{A} \cdot \bar{B}) = \bar{A} \times (\nabla \times \bar{B}) + \bar{B} \times (\nabla \times \bar{A}) + (\bar{A} \cdot \nabla)\bar{B} + (\bar{B} \cdot \nabla)\bar{A} \\

    &\nabla\cdot(\bar{A}\times \bar{B}) = \bar{B}\cdot(\nabla\times \bar{A}) - \bar{A}\cdot(\nabla\times \bar{B}) \\

    &\nabla\times(\bar{A}\times \bar{B}) = \bar{A}(\nabla\cdot \bar{B}) - \bar{B}(\nabla\cdot \bar{A}) + (\bar{B}\cdot \nabla)\bar{A} - (\bar{A}\cdot \nabla)\bar{B}
\end{align*}

\chapter{Electrostatics}
\section{Electric field}
The force on a test charge $(Q)$ from a single point charge $(q)$, a distance $\bar{\r}$ away, is calculated by Coulomb's law: 
\begin{align*}
    \bar{F} = \frac{1}{4 \pi \ep}\frac{q Q}{\r^2}\hat{\r}, \eq{2.1}
\end{align*}
where  $\ep = 8.85\ti{-12}\frac{\text{C}^2}{\text{N}\cdot \text{m}^2}$ is the permittivity of free space. 

If we have several point charges, at different distances from Q, we have an elektric field: 
\begin{align*}
    &\bar{F} = Q\bar{E}, \eq{2.3} \\
    &\bar{E} = \frac{1}{4\pi\ep}\sum_i \frac{q_i}{\r_i^2}\hat{\r_i}. \eq{2.4}
\end{align*}

If we have a continuous charge distribution, we can calculate the electric field by:
\begin{align*}
    \bar{E} = \frac{1}{4\pi\ep}\int \frac{1}{\r^2}\hat{\r}dq, \eq{2.5}
\end{align*}
where $dq$ is the charge in a small element, be it line charge $(dq=λdl)$, area charge $(dq=σda)$  or volume charge $(dq=ρdτ)$.

\subsubsection{Special cases:} 
$ $

Field of infinite length wire: \[ E = \frac{1}{4π\ep}\frac{2λ}{z}\]

Field of infinite plane: \[ \bar{E} = \frac{σ}{2\ep}\hat{a}\]

Field outside a uniformly charged sphere with total charge $q$, $r>R$: \[ \bar{E} = \frac{1}{4π\ep}\frac{q}{r^2}\hat{r}\]

\section{Divergence and Curl}
Field lines are imaginary lines used to represent the direction and strength of a field, such as an electric field. They start on positive charges and end on negative charges. The density of these lines indicates the field's strength; closer lines mean a stronger field.

Flux refers to the quantity of a field passing through a given surface. For an electric field, it is the product of the electric field strength and the area through which the field lines pass, considering the angle between the field lines and the surface.
\begin{align*}
    \Phi = \int_{\mathcal{S}} \bar{E} \cdot d\bar{a} \eq{2.11}
\end{align*}

Gauss's law states that the total electric flux through a closed surface is proportional to the charge enclosed within that surface.
\begin{align*}
    \oint \bar{E} \cdot d\bar{a} = \frac{Q_{\text{enc}}}{\ep}, \eq{2.13}
\end{align*}
where $Q_{\text{enc}}$ is the charge enclosed by the surface. 
It can be rewritten with the divergence thereom, when $Q_{\text{enc}} = \int_{\mathcal{V}}ρdτ$: 
\begin{align*}
    \nabla \cdot \bar{E} = \frac{ρ}{\ep}. \eq{2.14}
\end{align*} 

To calculate the curl of the electric fgield we see that: 
\begin{align*}
    \int_{\bar{a}}^{\bar{b}} \bar{E} \cdot d\bar{l} = \frac{1}{4π\ep} \br{\frac{q}{r_a}-\frac{q}{r_b}}, \eq{2.18}
\end{align*}
which on a closed interval gives us: 
\begin{align*}
    \oint \bar{E} \cdot d\bar{l} = 0, \eq{2.19} \\
    \nabla \times \bar{E} = \bar{0}. \eq{2.20}
\end{align*}

\section{Work and Energy}
If you want to move a test charge $(Q)$ from point $\bar{a}$ to point $\bar{b}$, 
you have to do work against the electric field. 
The work done is the potential difference between the two points: 
\begin{align*}
    W = Q\br[2]{V(\bar{b}) - V(\bar{a})}. \eq{2.38}
\end{align*}
So if you want to move the point from ininfity to a point $\bar{r}$, and you have the reference point in infinity: 
\begin{align*}
    W = QV(\bar{r}). \eq{2.39}
\end{align*}

The work to move a collection of charges is:
\begin{align*}
    W &= \frac{1}{4π\ep}\sum_{i=1}^{n}\sum_{j>i}^{n}\frac{q_iq_j}{\r_{ij}}, \eq{2.40}\\
    &= \frac{1}{8π\ep}\sum_{i=1}^{n}\sum_{j\neq i}^{n}\frac{q_iq_j}{\r_{ij}}, \eq{2.41} \\
    &= \frac{1}{2}\sum_{i=1}^{n}q_iV(\bar{r}_i). \eq{2.42}
\end{align*}

For continuous charge distributions:
\begin{align*}
    W = \frac{1}{2}\int V dq, \eq{2.43}
\end{align*}
where again, $dq$ is the charge in a small element, be it line charge $(dq=λdl)$, area charge $(dq=σda)$  or volume charge $(dq=ρdτ)$.

If we take (2.43) for volume over all space, we get: 
\begin{align*}
    W = \frac{\ep}{2}\int E^2 dτ. \eq{2.45}
\end{align*}

\subsubsection{Special cases}

Energy of Q spread uniformly throughout sphere of radius R: 
\begin{align*}
    W_{tot} &= W_{out}+W_{in} \\
    &= \frac{Q^2}{8π\ep R} + \frac{3Q^2}{20π\ep R}
\end{align*}

\section{Conductors}
A conducter is a material where the electrons are free to move around. The electric field inside a conductor is zero, and the charges are distributed on the surface, like metal.
The electric field inside a conductor is zero, and when it is placed in an electric field, 
the charges collect on the surface to create an electrical field inside the conducter, that cancels out the initial field. 
\begin{align*}

Induced charges on a conductor are always on the surface, and the electric field is always perpendicular to the surface.
For a conductor with a cavity, the electric field is zero inside the cavity, and the charges are on the surface of the cavity.
Outside the conductor, the electric field is the same as if the conductor was not there.

\section{Laplace's equation}
To easily calculate the potential in a region, we can use Laplace's equation:
\begin{align*}
    \nabla^2 V = 0. \eq{3.4}
\end{align*}

In one dimension the solution is:
\begin{align*}
    V(x) = ax + b, \eq{3.6}
\end{align*}

but for 2 dimensions it is not easy to calculate exactly, but you can caluculate $V(x,y)$ as the average of the potential around the point: 
\begin{align*}
    V(x,y) = \frac{1}{2πR}\oint_{circle}Vdl, 
\end{align*}
which might make no sence, but it can be used to calculate the charge distribution, when boundary conditions are known. 

In three dimensions the value of $V$ at point $\bar{r}$ is the average value of $V$ over a spherical surface of radius $R$ centered at $\bar{r}$: 
\begin{align*}
    V(\bar{r}) = \frac{1}{4πR^2}\oint_{\text{sphere}}Vda. 
\end{align*}

\textbf{First uniqueness theorem:} The solution to Laplace’s equation in 
some volume $\mathcal{V}$ is uniquely determined if $V$ is specified on the 
boundary surface $\mathcal{S}$. Meaning that it doesn't matter how you get your solution, 
if it satisfies Laplace’s equation, and the boundary values are correct, then the answer is correct. 

\textbf{Second uniqueness theorem:} In a volume $\mathcal{V}$ surrounded by 
conductors and containing a specified charge density $\rho$, the electric field 
is uniquely determined if the \textit{total charge} on each conductor is given. 
(The region as a whole can be bounded by another conductor, 
or else unbounded.)


\section{The Method of Images}
The method of images can be used to solve advanced electrostatic problems, 
by replacing conductors with fictitious charges that mimic the same boundary conditions. 

For example, we have a infinite grounded plate, with a point charge a distance $d$ above it. 
The boundary potential is that $V=0$ when $z=0$ and that $V \rightarrow 0$ far from the charge (Fig 3.10).
The same boundary conditions are satisfied if we have a point charge at $-d$ instead of the conductor (Fig 3.11). 
Therefore the potential for a pointcharge above an infinite grounded plate is: 
\begin{align*}
    V(x,y,z) = \frac{1}{4π\ep}\br[2]{\frac{q}{\sqrt{x^2+y^2+(z-d)^2}}-\frac{q}{\sqrt{x^2+y^2+(z+d)^2}}}. \eq{3.9}
\end{align*}

From this we can calculate the induced surface charge on the conductor: 
\begin{align*}
    σ(x,y) &= -\ep \frac{\dd V}{\dd z}|_{z=0} \\
    &= \frac{-qd}{2π(x^2+y^2+d^2)^{3/2}}, \eq{3.10}
\end{align*}
and the total charge: 
\begin{align*}
    Q = \int σda = -q. \eq{3.11}
\end{align*}

We can also use the method of images to find the force on the point charge due to the induction: 
\begin{align*}
    \bar{F} = -\frac{1}{4π\ep} \frac{q^2}{(2d)^2}\hat{z}, \eq{3.12}
\end{align*}
but the work will not be the same for the plane, and the point charge, 
so if we want to calculate the energy to move the charge from infinity to d as:
\begin{align*}
    W = \int_{\infty}^{d}\bar{F}\cdot d\bar{l}= -\frac{1}{4π\ep}\frac{q^2}{4d}. \eq{3.14}
\end{align*}

You can not put the image charge in the reagion where you are calculating the potential, 
so in example 3.2, where we have a point charge a distance a away from a sperical grounded conductor, 
the image charge is inside the sphere. 

This gives the potential: 
\begin{align*}
    V(\bar{r}) = \frac{1}{4π\ep}\br{\frac{q}{\r}+\frac{q'}{\r'}}, \eq{3.17}
\end{align*}
where $q'$ is the image charge, and $\r$ and $\r'$ is the distance from $q$ and $q'$.

To calculate the potential from a multipole we use:
\begin{align*}
    V(\bar{r}) = \frac{1}{4π\ep}\sum_{n=0}^{\infty}\frac{1}{r^{n+1}}\int(r')^nP_n(\cos(α))ρ(\bar{r}')dτ', \eq{3.95}
\end{align*}
where $P_l$ is the Legendre polynomial, $ρ$ is the total charge of the distrubution. 
$n = 0 $ for monopole, $1$ for dipole and so on. 

For monopole and dipole we get: 
\begin{align*}
    V_{mon}(\bar{r}) = \elcon \frac{Q}{r}, \eq{3.97} \\
    V_{dip}(\bar{r}) = \elcon \frac{\bar{p}\cdot \hat{r}}{r^2}, \eq{3.99}
\end{align*}
where $\bar{p}$ is the dipole moment of the distrubution: 
\begin{align*}
    \bar{p} = \int \bar{r}'ρ(\bar{r}')dτ'. \eq{3.98}
\end{align*}

The electric field of a dipole is: 
\begin{align*}
    \bar{E}_{dip}(r,θ) = \frac{p}{4π\ep r^3}\br{2\cos(θ)\hat{r}+\sin(θ)\hat{θ}}. \eq{3.103}
\end{align*}

\chapter{Electric Fields in Matter}
\section{Dielectrics}
If you put a neutrol atom in an electric field, it is either ripped apart, being ionized, 
or it reaches an equilibrium, where the force from the electric field and the force holding the atom toghether cancel eachother, 
making the atom polarized. This will cause the atom to have a dipolemoment $\bar{p}$, with the same direction as $\bar{E}$. 
\begin{align*}
    \bar{p} = α\bar{E} \eq{4.1}, 
\end{align*}
here $α$ is called the atomic polarizability, and is dependent on the atoms structure. 

If an atom was made by a center charge $q$ surrounded by as spherical charged cloud $-q$ with radius $a$, 
then the electric field we place the atom in, would move the nucleus a distance $d$ from the center. 
Now the electric field in equilibrium would be: 
\begin{align*}
    E &= \elcon \frac{qd}{a^3}, \quad \text{or} \quad p = qd = (4π\ep a^3)E, \\
    \Rightarrow α &= 4π\ep a^3 = 3\ep v, \eq{4.2}
\end{align*}
where $v$ is the volume of the atom. 
\bil{fig4.1}

For molecules it is not so easy, as the dipolemoment might not be the same direction as E, 
and therefor we get 9 polarizability constants: 
\begin{align*}
    \begin{cases}
        p_x = \alpha_{xx} E_x + \alpha_{xy} E_y + \alpha_{xz} E_z, \\
        p_y = \alpha_{yx} E_x + \alpha_{yy} E_y + \alpha_{yz} E_z, \\
        p_z = \alpha_{zx} E_x + \alpha_{zy} E_y + \alpha_{zz} E_z.
    \end{cases} \eq{4.3}
\end{align*}


\subsection{Alignment of polar molecules}
Some molecules have their own dipolemoment, such as water, which are called polar molecules. 
If placed in an uniform electric field, the force on the postive end, cancels out with the force on the negative end, 
but there will be a torque 
\begin{align*}
    \bar{N} = (\bar{r}_+ \times \bar{F}_+)+(\bar{r}_- \times \bar{F}_-) = \bar{p}\times \bar{E}. \eq{4.4}
\end{align*}
\bil{fig4.4}
$\bar{N}$ is in such a direction to line $\bar{p}$ parallel to $\bar{E}$. 

If the field is not uniform over det atom, which is rare, there is a net force on the atom 
\begin{align*}
    \bar{F} = \br{\bar{p}\cdot \nabla}\bar{E}. \eq{4.5}
\end{align*}

An ideal dipole is the atrifical limit $d\rightarrow 0$ and $q \rightarrow \infty$. 
The energy of an ideal dipolemoment is 
\begin{align*}
    U = -\bar{p}\cdot \bar{E}, \eq{4.6} 
\end{align*}
and the interaction energy between two ideal dipoles with displacement $\bar{r}$ is 
\begin{align*}
    U = \elcon \frac{1}{r^3}\br[2]{\bar{p}_1\cdot\bar{p}_2-3\br{\bar{p}_1\cdot \hat{r}}\br{\bar{p}_2\cdot \hat{r}}}. \eq{4.7}
\end{align*}

If we place a material made of nonpolar molecules in an electric field,  
it will induce a dipole moment in all the molecules. If we place a polar material, 
the field will turn all the molecules. We get the polarization which is 
\begin{align*}
    \bar{P} = \text{dipole moment per unit volume. } 
\end{align*}

\section{Field of a Polarized Object}
To calculate the field that arises from the induced dipoles in a material placed in an electric field, 
we first calculate the potential by integrating each dipole over det volume $\bar{p}=\bar{P}dτ'$: 
\begin{align*}
    V(\bar{r}) &= \elcon \int_{\mathcal{V}} \frac{\bar{P}(\bar{r}')\cdot \hat{\r}}{\r^2}dτ', \eq{4.9} \\
    &= \elcon \oint_{\mathcal{S}} \frac{σ_b}{\r}da'+\elcon \int_{\mathcal{V}}\frac{ρ_b}{\r}dτ', \eq{4.13}
\end{align*}
where we have the surface charge density $σ_b = \bar{P}\cdot\hat{n}$ and the volume charge density $ρ_b = -\bar{\nabla}\cdot \bar{P}$. 

For a uniformly polarized sphere of radius R, the volume charge density is zero bacause $\bar{P}$ is uniform, but 
\begin{align*}
    σ_b = \bar{P}\cdot \hat{n} = P\cos θ,
\end{align*}
from this we get
\begin{align*}
    V(r,θ) = \begin{cases}
        \frac{P}{3\ep}r\cos θ & \text{for } r \leq R, \\
        \frac{P}{3\ep}\frac{R^3}{r^2}\cos θ & \text{for } r \geq R.
    \end{cases}
\end{align*}

So the field inside the sphere is uniform, and outside the sphere it behaves like a ideal dipole, 
whose dipole moment is the total dipole moment of a sphere $\bar{p} = \frac{4}{3}πR^3\bar{P}$.


\section{Electric Displacement}
Along with the bound charge, there is also fee charge which could be electrons on a fonductor, 
or ions embedded in the dielectric material, so any charge not assosiated with $\bar{P}$. 
Within the dielectric, the total charge density is 
\begin{align*}
    ρ = ρ_b + ρ_f, \eq{4.20}
\end{align*}
and Gauss's law is 
\begin{align*}
    \ep \bar{\nabla}\cdot\bar{E} = ρ = ρ_b + ρ_f = -\bar{\nabla}\cdot\bar{P}+ρ_f, 
\end{align*}
where $\bar{E}$ is the toral field. 

This lead to a term for $ρ_f$: 
\begin{align*}
    ρ_f &= \bar{\nabla}\cdot\br{\ep\bar{E}+\bar{P}}, \\
    &= \bar{\nabla}\cdot\bar{D}, \eq{4.22} \\
    \Rightarrow \oint &\bar{D}\cdot d\bar{a} = Q_{f_{enc}}. \eq{4.23}
\end{align*}
Here $\bar{D}$ is known as the electric displacement. 

Typically we know $ρ_f$, and we can therefor calculate D. When computing the displacement look for symmetry, 
if there is spherical, cylindircal or plane symmetri we can use (\ref{eq4.23}) because $\bar{\nabla}\times\bar{D} = 0$. 

\section{Linear Dielectrics}
The polarization is proportional to the electric field: 
\begin{align*}
    \bar{P} = \ep Χ_e \bar{E} \eq{4.30}, 
\end{align*}
where $χ_e$ is the electric susceptibility, which is dependant on the structure of the material. 
Materials that obey eq. 4.30 are called linear dielectrics. 
Notice that $\bar{E}$ is the total electric field, and therefore we can not calculate $\bar{P}$ from eq. 4.30, 
instead we could calculate 
\begin{align*}
    \bar{D} = \ep \bar{E}+\bar{P} = \ep(1+χ_e)\bar{E}=\epsilon \bar{E}, \eq{4.31}
\end{align*}
where we have the permittivity of the material, and dielectric constant:
\begin{align*}
    \epsilon = \ep(1+χ_e), \eq{4.33} \\
    \epsilon_r = 1+χ_e = \frac{\epsilon}{\ep}. \eq{4.34}
\end{align*}

\bil{fig4.2}

When we have a charge distrubution in homogeneous linear dielectric, the field is calculated as in vacum, but using $\epsilon$ instead of $\ep$. 
To calculate the full energy stored in a electrostatic system in a linear dielectric, we rewrite (\hyperref[eq2.45]{2.45}) to
\begin{align*}
    W = \frac{\ep}{2}\int \epsilon_r E^2dτ = \frac{1}{2}\int \bar{D}\cdot \bar{E}dτ. \eq{4.58}
\end{align*}
