\chapter{Gaussian Method}
Using Gaussian methods for the variation principle and correlated gaussian method, 
we define the wavefunction as a sum of gaussians (denoted $\ket{g}$). 
\begin{align*}
    ψ(\v{r}) = \sum_{i=1}^{n_g} c_i\ket{g_i} = \braket{\v{r}}{g}. 
\end{align*}

By using this we get the Schrödinger equation as 
\begin{align*}
    H\ket{ψ} &= E\ket{ψ}, \\
    \mathcal{H}c &= E\mathcal{N}c, 
\end{align*}
where 
\begin{align*}
    \mathcal{H} &= \br[4]{H_{i,j}} = \bra{g_i}\hat{H}\ket{g_j}, \\
    \mathcal{N} &= \br[4]{N_{i,j}} = \braket{g_i}{g_j}.
\end{align*}

\plainbreak{1}
The correlated gaussian is defined by: 
\begin{align*}
    \braket{\v{r}}{g} &= \text{exp}\br{-\sum_{i,j=1}^{N}A_{i,j}\v{r}_i\cdot \v{r_j}+\sum_{i=1}^{N}\v{s}_i\cdot \v{r}_i}, \\
    &= \e{-\v{r}^TA\v{r}+\v{s}^T\v{r}}. 
\end{align*}
which gives the notation
\begin{align*}
    \ket{g} = \ket{A, s}.
\end{align*}

The overlap of these gaussians is given by \cite{ana}
\begin{align*}
    \braket{g'}{g} = \br{\frac{π^N}{\text{det}B}}\e{\frac{1}{4}\v{v}^TB^{-1}\v{v}} \equiv M, 
\end{align*}
where $B = A+A'$ and $\v{v} = \v{s} + \v{s}'$. 



\plainbreak{1}
To find the fouriertransformed correlated gaussian we evaluate the intergral: 
\begin{align*}
    \braket{\v{p}}{g} &= \int d^{3N}x\e{\v{x}^T\v{p}} \e{-\v{r}^TA\v{r}+\v{s}^T\v{r}},\\
    &= \br{\frac{π^N}{\text{det}(A)}}^{3/2} \e{\frac{1}{4}(\v{s}-i\v{p})^TA^{-1}(\v{s}-i\v{p})}, \\
    &= \br{\frac{π^N}{\text{det}(A)}}^{3/2} \e{\frac{1}{4}\v{s}^TA^{-1}\v{s}} \cdot \e{-\frac{1}{4}\v{p}^TA^{-1}\v{p}} \cdot \e{-\frac{1}{2}i\v{p}^TA^{-1}\v{s}}, 
\end{align*}
which gives the fourier transformed gaussian
\begin{align*}
    \ket{\f{g}} = \e{-\v{p}^T\f{A}\v{p}+\f{\v{s}}^T\v{p}},
\end{align*}
where $\f{A}=\frac{1}{4}A^{-1}$ and $\f{\v{s}}=\frac{-i}{2}A^{-1}\v{s}$. 
And the $\mathcal{M}$ function is introduced 
\begin{align*}
    \M (A,\v{s}) = \br{\frac{π^N}{\text{det}(A)}}^{3/2} \e{\frac{1}{4}\v{s}^TA^{-1}\v{s}},
\end{align*}
giving us the final solution 
\begin{align*}
    \braket{\v{p}}{g} = \M(A,\v{s}) \ket{\f{g}}.
\end{align*}


The overlap in momentum space can be calculated as 
\begin{align*}
    \braket{g'}{g} &= \int \frac{d^{3N}p}{(2π)^{3N}} \braket{g'}{p} \braket{p}{g}, \\
    &= \br{\frac{π^N}{\text{det}A'}\frac{π^N}{\text{det}A}}^{3/2} \frac{\e{\frac{1}{4}\v{s}'^TA'^{-1}\v{s}'}\e{\frac{1}{4}\v{s}^TA^{-1}\v{s}}}{(2π)^{3N}}\braket{\f{g}'}{\f{g}}, \\
    &= \M(A',\v{s}')\M(A,\v{s})\frac{\braket{\f{g}'}{\f{g}}}{(2π)^{3N}}.
\end{align*}

We can now use the overlap to calculate the matrix element for the general momentum operator $f(c^T\v{p})$.
\begin{align*}
    \bra{g'}f(c^T\v{p})\ket{g} &= \int \frac{d^{3N}p}{(2π)^{3N}}\braket{g'}{\v{p}}f(c^T\v{p})\braket{\v{p}}{g}, \\
    &= \M(A',\v{s}')\M(A,\v{s}) \int \frac{d^{3N}p}{(2π)^{3N}} \bra{\f{g}'}f(c^T\v{x})\ket{\f{g}}, 
\end{align*}
using equation (\ref{place}) we continue
\begin{align*}
    &= \M(A',\v{s}')\M(A,\v{s}) \braket{\f{g}'}{\f{g}} \br{\frac{γ}{π}}^{3/2}2π\frac{\e{-γ(iη)^2}}{iγη}\int xdx \f{f}(x)\e{-γx^2}\sinh(2iγηx), \\
    &= \braket{g'}{g}\br{\frac{γ}{π}}^{3/2}2π\frac{\e{γη^2}}{γη}\int xdx \f{f}(x)\e{-γx^2}\sin(2γηx)
\end{align*}

\chapter{W-operator}
the coupeling operator $W$ is given by 
\begin{align*}
    W = (\v{τπ})(\v{σr})f(r), 
\end{align*}
where $(\v{τπ})$ desides the isospin of the nucleon and the resulting pion, and is given by 
\begin{align*}
    (\v{τπ}) = τ_0π^0+\sqrt{2}τ_-π^++\sqrt{2}τ_+π^-,
\end{align*}
where $τ$ are the Pauli spin matrices. This gives us the possible states that a bare deuterium nucleus can be in: 
\begin{align*}
    (\v{τπ})\ket{pn} &= (τ_0π^0+\sqrt{2}τ_-π^++\sqrt{2}τ_+π^-)\ket{p} + (τ_0π^0+\sqrt{2}τ_-π^++\sqrt{2}τ_+π^-)\ket{n}, \\
    &= (π^0\ket{p}+\sqrt(2)π^+\ket{n})+(-π^0\ket{n}+\sqrt{2}π^-\ket{n}) 
\end{align*}
